Map initialization const map = L.map('map').setView([39.5, -98.5], 4); L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', { attribution: '© OpenStreetMap contributors' }).addTo(map); // 2. Clusters const damCluster = L.markerClusterGroup(); const nwmCluster = L.markerClusterGroup(); map.addLayer(damCluster); map.addLayer(nwmCluster); // 3. Globals let allDamFeatures = []; // 4. Styling function getHazardColor(h) { if (h === "High") return "red"; if (h === "Significant") return "orange"; if (h === "Low") return "green"; return "gray"; } // 5. Filters function getSelectedHazards() { return Array.from(document.querySelectorAll('.filter-box input:checked')) .map(cb => cb.value); } // 6. Forecast (FIXED LOCATION) function loadForecastForDam(feature) { const props = feature.properties; const comid = props.comid || props.COMID; if (!comid) { alert("No COMID available"); return; } const url = `https://api.water.noaa.gov/nwps/v1/gauges/${comid}/forecast`; fetch(url) .then(r => r.json()) .then(data => { const times = data.forecast?.timeSeries?.map(d => d.validTime) || []; const flows = data.forecast?.timeSeries?.map(d => d.value) || []; const modal = document.getElementById('forecastModal'); const statusDisplay = document.getElementById('statusDisplay'); modal.style.display = "block"; statusDisplay.innerHTML = `

${props.name || "Dam"}

COMID: ${comid}

`; const ctx = document.getElementById('forecastChart').getContext('2d'); if (window.forecastChart) window.forecastChart.destroy(); window.forecastChart = new Chart(ctx, { type: 'line', data: { labels: times, datasets: [{ label: 'Streamflow (cfs)', data: flows, borderWidth: 2 }] } }); }); } // 7. Render dams (FIXED CLICK) function renderDams() { damCluster.clearLayers(); const selected = getSelectedHazards(); allDamFeatures.forEach(feature => { if (!selected.includes(feature.properties.hazard)) return; const [lon, lat] = feature.geometry.coordinates; const marker = L.circleMarker([lat, lon], { radius: 6, fillColor: getHazardColor(feature.properties.hazard), color: "#fff", weight: 1, fillOpacity: 0.8 }); marker.bindTooltip(feature.properties.name || "Dam"); // ✅ CLICK HANDLER (this was missing before) marker.on('click', () => loadForecastForDam(feature)); damCluster.addLayer(marker); }); } // 8. Load GeoPackage fetch('dams.gpkg') .then(r => r.arrayBuffer()) .then(b => geopackage.open(b)) .then(gpkg => { gpkg.getFeatureTables().forEach(table => { const features = gpkg.iterateGeoJSONFeatures(table); for (let f of features) allDamFeatures.push(f); }); renderDams(); }); // 9. Load NWM function loadNWM() { fetch("https://api.water.noaa.gov/nwps/v1/gauges") .then(r => r.json()) .then(data => { nwmCluster.clearLayers(); data.forEach(site => { if (!site.latitude) return; const marker = L.circleMarker([site.latitude, site.longitude], { radius: 5, fillColor: "blue", color: "#fff", fillOpacity: 0.7 }); marker.bindTooltip(`${site.name}
Flow: ${site.flow}`); nwmCluster.addLayer(marker); }); }); } loadNWM(); setInterval(loadNWM, 600000); // 10. Hook filters document.querySelectorAll('.filter-box input') .forEach(cb => cb.addEventListener('change', renderDams));
